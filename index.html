<script>
    let players = [];
    let selectedCategories = [];
    const gameData = {
        "کار": ["پزیشک", "ئەندازیار", "مامۆستا", "پۆلیس", "جوتیار", "نۆژەنکەرەوە"],
        "خواردن": ["پیتزا", "کباب", "دۆڵمە", "بەرگر", "برنج", "شێلم"],
        "کەلوپەل": ["قەڵەم", "مۆبایل", "کۆمپیوتەر", "سەعات", "کورسی", "تەلەفزیۆن"],
        "ئاژەڵ": ["شێر", "پشیلە", "فیل", "ئەسپ", "سەگ", "ورچ"],
        "شار": ["هەولێر", "سلێمانی", "دۆهۆک", "کەرکوک", "هەڵەبجە", "بەغداد"],
        "هەمەجۆر": ["مانگ", "خۆر", "دەریا", "چیا", "کتێب", "فڕۆکە"]
    };

    let secretWord = "";
    let imposters = [];
    let turnIndex = 0;
    let isRevealed = false;
    let timerInterval;

    function nextPage(p) {
        document.querySelectorAll('.page').forEach(page => page.classList.remove('active'));
        document.getElementById('page' + p).classList.add('active');
    }

    function addPlayerName() {
        const input = document.getElementById('nameInput');
        const name = input.value.trim();
        if (name && players.length < 10) {
            players.push(name);
            renderNames();
            input.value = "";
        }
    }

    function removeName(index) {
        players.splice(index, 1);
        renderNames();
    }

    function renderNames() {
        const container = document.getElementById('namesContainer');
        container.innerHTML = players.map((name, index) => `
            <div class="name-tag">
                <b>${name}</b>
                <span class="remove-btn" onclick="removeName(${index})">×</span>
            </div>
        `).join('');
    }

    function toggleCat(el) {
        const cat = el.getAttribute('data-cat');
        if (selectedCategories.includes(cat)) {
            selectedCategories = selectedCategories.filter(c => c !== cat);
            el.classList.remove('selected');
        } else {
            selectedCategories.push(cat);
            el.classList.add('selected');
        }
    }

    function startGame() {
        if (players.length < 3) return alert("لایەنی کەم ٣ یاریزان زیاد بکە");
        if (selectedCategories.length === 0) return alert("تکایە بەشێک یان زیاتر هەڵبژێرە");

        let allWords = [];
        selectedCategories.forEach(cat => allWords = allWords.concat(gameData[cat]));
        secretWord = allWords[Math.floor(Math.random() * allWords.length)];

        let count = parseInt(document.getElementById('imposterCount').value) || 1;
        imposters = [];
        let tempPlayers = [...players];
        for(let i=0; i<count; i++) {
            let idx = Math.floor(Math.random() * tempPlayers.length);
            imposters.push(tempPlayers.splice(idx, 1)[0]);
        }

        turnIndex = 0;
        showTurn();
        nextPage(3);
    }

    function showTurn() {
        isRevealed = false;
        document.getElementById('currentPlayerName').innerText = players[turnIndex];
        const card = document.getElementById('roleCard');
        card.innerText = "بۆ بینینی ڕۆڵەکەت کلیک بکە";
        card.style.background = "rgba(255, 255, 255, 0.95)";
        card.style.color = "#1a0b2e";
        document.getElementById('nextTurnBtn').style.display = "none";
    }

    function revealRole() {
        if (isRevealed) return;
        const card = document.getElementById('roleCard');
        const pName = players[turnIndex];

        if (imposters.includes(pName)) {
            card.innerText = "تۆ IMPOSTER یت!";
            card.style.background = "var(--imposter-red)";
            card.style.color = "white";
            card.style.boxShadow = "0 0 30px var(--imposter-red)";
        } else {
            card.innerText = "وشەکە: " + secretWord;
            card.style.background = "#2e7d32";
            card.style.color = "white";
        }

        isRevealed = true;
        document.getElementById('nextTurnBtn').style.display = "inline-block";
    }

    function nextTurn() {
        turnIndex++;
        if (turnIndex < players.length) {
            showTurn();
        } else {
            startTimer();
        }
    }

    function startTimer() {
        nextPage(4);
        let seconds = parseInt(document.getElementById('gameTime').value) || 60;
        const display = document.getElementById('timerValue');

        timerInterval = setInterval(() => {
            let mins = Math.floor(seconds / 60);
            let secs = seconds % 60;
            display.innerText = `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;

            if (seconds <= 0) {
                clearInterval(timerInterval);
                finishGame();
            }
            seconds--;
        }, 1000);
    }

    function stopTimerAndFinish() {
        clearInterval(timerInterval);
        finishGame();
    }

    function finishGame() {
        document.getElementById('finalWord').innerText = secretWord;
        document.getElementById('finalImposters').innerText = imposters.join(" ، ");
        nextPage(5);
    }

    function resetGame() {
        players = [];
        selectedCategories = [];
        document.getElementById('namesContainer').innerHTML = "";
        document.querySelectorAll('.category-item').forEach(i => i.classList.remove('selected'));
        nextPage(1);
    }
</script>
